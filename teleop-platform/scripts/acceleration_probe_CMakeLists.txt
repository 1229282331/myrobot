cmake_minimum_required(VERSION 3.22)
project(e6_acceleration_probe)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(parameter_traits REQUIRED)
find_package(osqp REQUIRED)
find_library(ACCELERATION_PLUGIN moveit_acceleration_filter PATHS /opt/ros/jazzy/lib REQUIRED)
find_package(srdfdom REQUIRED)
find_package(urdf REQUIRED)
add_executable(probe probe_acceleration_filter.cpp)
ament_target_dependencies(probe moveit_core rclcpp rclcpp_lifecycle srdfdom urdf)
target_link_libraries(probe parameter_traits::parameter_traits osqp::osqp ${ACCELERATION_PLUGIN})
target_compile_features(probe PUBLIC cxx_std_17)
