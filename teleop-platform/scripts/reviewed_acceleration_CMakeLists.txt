cmake_minimum_required(VERSION 3.22)
project(e6_servo_filter_patch)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(parameter_traits REQUIRED)
find_package(osqp REQUIRED)
find_package(pluginlib REQUIRED)
find_package(srdfdom REQUIRED)
find_package(urdf REQUIRED)
add_library(e6_acceleration_filter SHARED reviewed_acceleration_filter.cpp)
ament_target_dependencies(e6_acceleration_filter moveit_core rclcpp rclcpp_lifecycle srdfdom urdf pluginlib)
target_link_libraries(e6_acceleration_filter parameter_traits::parameter_traits osqp::osqp)
target_compile_features(e6_acceleration_filter PUBLIC cxx_std_17)
add_executable(probe_reviewed probe_acceleration_filter.cpp)
target_compile_definitions(probe_reviewed PRIVATE E6_REVIEWED_FILTER)
ament_target_dependencies(probe_reviewed moveit_core rclcpp rclcpp_lifecycle srdfdom urdf)
target_link_libraries(probe_reviewed e6_acceleration_filter parameter_traits::parameter_traits osqp::osqp)
target_compile_features(probe_reviewed PUBLIC cxx_std_17)
pluginlib_export_plugin_description_file(moveit_core plugins.xml)
install(TARGETS e6_acceleration_filter LIBRARY DESTINATION lib)
ament_package()
