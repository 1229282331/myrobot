cmake_minimum_required(VERSION 3.22)
project(e6_workspace_probe)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(srdfdom REQUIRED)
find_package(urdf REQUIRED)
add_executable(workspace_probe workspace_probe.cpp)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/held_retreat.cpp")
  target_sources(workspace_probe PRIVATE held_retreat.cpp)
endif()
ament_target_dependencies(workspace_probe moveit_core srdfdom urdf)
